\usepackage{caption}

%----------------------------------------------------------------------------------------
%   COMMON
%----------------------------------------------------------------------------------------

\DeclareCaptionFont{fg}{\color{fg}}
\DeclareCaptionFont{accent}{\color{accent}}

%----------------------------------------------------------------------------------------
%   CODE: LSTLISTING
%----------------------------------------------------------------------------------------

\DeclareCaptionFormat{listing}{\colorbox{border}{\parbox{\dimexpr\linewidth-2\fboxsep\relax}{#1#2#3}}}
\captionsetup[lstlisting]{
  format=listing,
  labelfont=accent,
  textfont=fg,
  labelformat=default,
  listformat=empty,
  singlelinecheck=false,
  margin=0pt,
  font={bf,footnotesize}
}

%----------------------------------------------------------------------------------------
%   TABLES
%----------------------------------------------------------------------------------------

\captionsetup[table]{
  labelformat=default,
  listformat=empty,
}

%----------------------------------------------------------------------------------------
%   FIGURES
%----------------------------------------------------------------------------------------


\captionsetup[figure]{
	labelformat=default,
	listformat=empty,
	labelfont={bf,footnotesize},  % Bold and small label like lstlisting
	textfont=footnotesize,        % Small text
	font=footnotesize             % Overall small font
}


\captionsetup[table]{
	labelformat=default,
	listformat=empty,
	labelsep=period,
	labelfont={bf,footnotesize},
	textfont=footnotesize,
	justification=centering
}
