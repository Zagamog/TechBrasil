\usepackage{hyperref}
\usepackage{bookmark}
\usepackage[all]{hypcap} % moves the ref jumping points to the top of images (instead of the caption)

%----------------------------------------------------------------------------------------
%   HYPERLINKS IN THE DOCUMENTS
%----------------------------------------------------------------------------------------
\hypersetup{
	hidelinks=false,        % Changed to show colored links
	backref=true,
	pagebackref=true,
	hyperindex=true,
	colorlinks=true,
	breaklinks=true,
	urlcolor=blue,          % Standard blue for URLs
	linkcolor=black,        % Internal links stay black
	citecolor=green,        % Citations in green
	bookmarks=true,
	bookmarksopen=false,
	pdftitle={\TITLE},
	pdfauthor={\AUTHOR},
	pdfsubject={\SUBJECT}}

% Alternative color options:
% urlcolor=teal,        % Teal is also standard
% urlcolor=purple,      % Purple is another option
% urlcolor=red,         % Red if you want it to stand out

\bookmarksetup{
	open,
	numbered,
	addtohook={%
		\ifnum\bookmarkget{level}=0 % chapter
		\bookmarksetup{bold}%
		\fi
		\ifnum\bookmarkget{level}=-1 % part
		\bookmarksetup{color=accent,bold}%
		\fi
	}
}
